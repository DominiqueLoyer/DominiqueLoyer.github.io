# [8/9] Fonction de prétraitement
from nltk.corpus import stopwords
from nltk.stem import WordNetLemmatizer
from nltk.tokenize import word_tokenize
stop_words_set = set(stopwords.words('english'))
lemmatizer     = WordNetLemmatizer()

def preprocess_text(text):
    if not isinstance(text, str): return ""
    tokens = word_tokenize(text.lower())
    return ' '.join(lemmatizer.lemmatize(w) for w in tokens if w.isalpha() and w not in stop_words_set)

# [9/9] Topics
AP_TAR_PATH  = os.path.join(DRIVE_PROJECT_PATH, "AP.tar")
TOPICS_DIR   = os.path.join(DRIVE_PROJECT_PATH, "Topics-requetes")
QRELS_DIR    = os.path.join(DRIVE_PROJECT_PATH, "jugements de pertinence")
K_RESULTS    = 1000

def parse_topics(file_path):
    topics = {}
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()
    for m in re.finditer(r"(.*?)", content, re.DOTALL):
        tc = m.group(1)
        num = re.search(r"\s*Number:\s*(\d+)", tc, re.IGNORECASE)
        if not num: continue
        tid   = num.group(1).strip()
        title = (re.search(r"\s*(.*?)\s*(?=<|$)", tc, re.IGNORECASE|re.DOTALL) or [None,''])[1] if re.search(r"<title>", tc) else ''
        desc  = (re.search(r"<desc>\s*Description:\s*(.*?)\s*(?=<|$)", tc, re.IGNORECASE|re.DOTALL) or [None,''])[1] if re.search(r"<desc>", tc) else ''
        if tid: topics[tid] = {'title': title.strip(), 'desc': desc.strip()}
    return topics

topic_files = sorted(glob.glob(os.path.join(TOPICS_DIR, "topics.*.txt")))
all_topics  = {}
for tf in topic_files: all_topics.update(parse_topics(tf))

queries_short = {qid: d['title'] for qid,d in all_topics.items()}
queries_long  = {qid: d['title']+" "+d['desc'] for qid,d in all_topics.items()}
queries_short_preprocessed = {qid: preprocess_text(q) for qid,q in queries_short.items()}
queries_long_preprocessed  = {qid: preprocess_text(q) for qid,q in queries_long.items()}

print(f"\n✔ {len(all_topics)} topics chargés.")
print("--- Configuration Complète Terminée ---")
